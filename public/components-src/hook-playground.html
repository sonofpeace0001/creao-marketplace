Type fast. The debounced value only updates once you stop for the chosen delay.

useDebounce.tsexport function useDebounce(value, delay) { const [v, setV] = useState(value); useEffect(() => { const id = setTimeout(() => setV(value), delay); return () => clearTimeout(id); }, [value, delay]); return v; } // Settles after 500ms of quiet
